## 13. Item — 道具数量系统

命名空间 `Interface.Item`

```csharp
public class ItemChangeData          // 数量变更可变参数对象
{
    public int ID;                    // 道具标识
    public int Count;                 // AddItem/MinusItem 用正请求量；SetItem 用非负目标数量
    public ChangeMethod Method { get; set; }
}
public class ChangeMethod { public string Method { get; set; } }

public struct ItemStatus             // 只读状态快照；状态码由上层解释
{
    public int Status { get; }        // 上层解释的状态码
    public int ItemId { get; }
    public int Count { get; }         // 真实数量（时长只写 Time，不写 Count）
    public int Time { get; }          // 状态剩余秒数，无持续为 0
    // 构造：ItemStatus(int itemId, int status, int count, int time)
}

public interface IItemSystem
{
    void AddItem(ItemChangeData changeData);                 // 加
    bool MinusItem(ItemChangeData changeData);               // 扣；完整满足返回 true，否则 false 且不扣
    void SetItem(ItemChangeData changeData);                 // 设为指定非负值
    int GetCount(int id);                                    // 不存在返回 0
    void AddMax(ItemChangeData changeData);                  // 调整上限，正加负减
    int CdTime(int id, out int cdCfg, out int changeCountCfg); // 距下次恢复秒数 + 单次恢复量
    int GetMax(int id);                                      // 无上限返回 0
    bool IsMax(int id);
    bool CanUse(int id, int count);
    ItemStatus GetItemStatus(int itemId);
    void AddChanged(int itemId, Action<ItemStatus> changed);    // 订阅
    void RemoveChanged(int itemId, Action<ItemStatus> changed); // 退订
}
```

## 16. Flow — 流程调度

命名空间 `Interface.Flow`

```csharp
public static class FlowGroup { public const string Dynamic = "Dynamic"; }
public enum FlowInsertPosition { First, Last, AfterCurrent }
public enum FlowExecutorState { Idle, Running, Stopping }

public readonly struct FlowToken
{
    public readonly string Uid;
    public readonly string Group;
    public readonly int Priority;
    public bool IsValid => !string.IsNullOrEmpty(Uid);
    // 构造：FlowToken(string uid, string group, int priority)
}
public readonly struct FlowContext { public readonly string Group; }  // 构造：FlowContext(string group)

public interface IFlowExecutor
{
    FlowExecutorState State { get; }
    void Invoke(FlowContext context);
    void Stop();
    void TokenFinished(FlowToken token);
    void Insert(IFlowItem item, FlowInsertPosition position);
    void Remove(IFlowItem item);
}
public interface IFlowScheduler { void Enqueue(IFlowItem item, FlowInsertPosition position); }

public interface IFlowItem
{
    string Group { get; }
    int Priority { get; }
    bool DeleteAfterExecute { get; }
    bool CanRun(FlowContext context);   // 必须无副作用、不依赖调用次数（FlowSystem 会重复调用预判）
    void Run(FlowContext context, FlowToken token, IFlowExecutor executor);
}
public interface IFlowUserSkipPolicy
{
    bool CanUserSkip(FlowContext context);   // 只控制主界面输入，不跳过/停止/取消 FlowItem
}
public interface ICancellableFlowItem
{
    void Cancel();   // 执行前被移除或执行中被停止时收到；必须重复调用安全
}

public interface IFlowSystem
{
    bool CanUserSkip { get; }                     // 当前及后续项是否都允许玩家操作主界面
    event Action<bool> CanUserSkipChanged;         // 仅在值实际变化时触发
    void Register(IFlowItem item);
    void UnRegister(IFlowItem item);
    IReadOnlyList<IFlowItem> GetItemsForGroup(string group);
}
```

## 17. Tutorial — 新手引导

命名空间 `Interface.Tutorial`

```csharp
public interface ITutorialSystem
{
    IDisposable Register(ITutorialProgram program);
    void Notify(ITutorialSignal signal);
}

public interface ITutorialProgram
{
    string Key { get; }
    void CollectCandidates(ITutorialSignal signal, ICollection<TutorialStep> output);
    bool TryGetNext(TutorialStepKey current, out TutorialStep next);
}
public interface ITutorialSignal { }                     // 空标记接口
public interface ITutorialLifetime : IDisposable
{
    void Suspend();
    void Resume();
}
public interface ITutorialGroupLifetime : ITutorialLifetime { }  // 归引导组所有，跨同组步骤复用

public interface ITutorialStepScope
{
    ITutorialSignal TriggerSignal { get; }
    bool IsCurrent { get; }
    void Add(ITutorialLifetime lifetime);
    T GetOrAddGroupLifetime<T>(Func<T> factory) where T : class, ITutorialGroupLifetime;
    bool TryGetGroupLifetime<T>(out T lifetime) where T : class, ITutorialGroupLifetime;
    void SetPrepared<T>(T prepared) where T : class;
    bool TryGetPrepared<T>(out T prepared) where T : class;
    void Ready();
    void Fail(string reason);
    void Notify(ITutorialSignal signal);
}

public enum TutorialStepOutcome { Completed, Interrupted, Released, Failed }

public readonly struct TutorialStepKey : IEquatable<TutorialStepKey>
{
    public string ProgramKey { get; }
    public int GroupId { get; }     // 构造时校验 > 0
    public int Order { get; }       // 构造时校验 > 0
    public bool IsValid => !string.IsNullOrWhiteSpace(ProgramKey) && GroupId > 0 && Order > 0;
    public bool IsSameGroup(TutorialStepKey other);   // ProgramKey + GroupId 相等
    // 构造：TutorialStepKey(string programKey, int groupId, int order)
    // 实现了 Equals / GetHashCode / ToString("{ProgramKey}/{GroupId}/{Order}") / == / !=
}

public readonly struct TutorialRank : IEquatable<TutorialRank>, IComparable<TutorialRank>
{
    public int Priority { get; }
    public bool IsValid { get; }         // kind ∈ [1..4]
    public bool IsMain { get; }          // kind = 1 或 2
    public bool IsMainWeak { get; }      // kind = 1
    public bool IsMainEnforce { get; }   // kind = 2
    public bool IsTile { get; }          // kind = 3
    public bool IsLevelEntry { get; }    // kind = 4
    public static TutorialRank MainWeak(int priority);
    public static TutorialRank MainEnforce(int priority);
    public static TutorialRank Tile(int priority);
    public static TutorialRank LevelEntry(int priority);
    // CompareTo：先比 kind，再比 priority（数值小 = 优先级高）
}

public sealed class TutorialStep
{
    public TutorialStepKey Key { get; }
    public TutorialRank Rank { get; }
    public int DelayMs { get; }
    public Func<ITutorialSignal, bool> MatchesEnd { get; }
    public Action<ITutorialStepScope> Prepare { get; }
    public IReadOnlyList<Func<ITutorialStepScope, ITutorialLifetime>> BeginActions { get; }
    public Func<ITutorialStepScope, ITutorialLifetime> Present { get; }
    public IReadOnlyList<Action<ITutorialStepScope>> EndActions { get; }
    public Action<TutorialStepOutcome> OnOutcome { get; }
    public IReadOnlyList<string> BindingNames { get; }
    // 构造：TutorialStep(key, rank, delayMs=0, matchesEnd=null, prepare=null,
    //         beginActions=null, present=null, endActions=null, onOutcome=null, bindingNames=null)
    // 校验：key.IsValid 与 rank.IsValid 必须为 true，delayMs ≥ 0
}
```
